async function buscarPacientePorProntuario() { const prontuarioInput = document.getElementById('input-prontuario').value.trim(); const infoBox = document.getElementById('paciente-info'); const novoBox = document.getElementById('novo-paciente-container'); dataNascimentoPacienteAtual = ""; if (!prontuarioInput) { if (infoBox) infoBox.style.display = 'none'; if (novoBox) novoBox.style.display = 'none'; return; } const dataAtual = new Date(); const ano = dataAtual.getFullYear(); const mes = String(dataAtual.getMonth() + 1).padStart(2, '0'); const dia = String(dataAtual.getDate()).padStart(2, '0'); const hojeStr = `${ano}-${mes}-${dia}`; let pacienteEncontrado = null; let setorEncontrado = null; let leitoEncontrado = null; let jaCadastradoHoje = false; // 1. Procura primeiro no plantão de hoje const { data: plantaoAtual } = await _supabase .from('plantoes') .select('dados_json') .eq('data_chave', hojeStr) .maybeSingle(); if (plantaoAtual && plantaoAtual.dados_json && Array.isArray(plantaoAtual.dados_json)) { const pHoje = plantaoAtual.dados_json.find(p => p.prontuario && p.prontuario.toString().trim() === prontuarioInput); if (pHoje) { pacienteEncontrado = pHoje; setorEncontrado = pHoje.setor; leitoEncontrado = pHoje.leito || "01"; dataNascimentoPacienteAtual = pHoje.dtNasc || ""; jaCadastradoHoje = true; } } // 2. Se não achou hoje, varre o histórico de plantões anteriores (dias atrás) if (!pacienteEncontrado) { const { data: historicoPlantoes } = await _supabase .from('plantoes') .select('dados_json') .order('data_chave', { ascending: false }) .limit(30); // Busca nos últimos 30 dias de histórico if (historicoPlantoes && historicoPlantoes.length > 0) { for (const plantao of historicoPlantoes) { if (plantao.dados_json && Array.isArray(plantao.dados_json)) { const pAntigo = plantao.dados_json.find(p => p.prontuario && p.prontuario.toString().trim() === prontuarioInput); if (pAntigo) { pacienteEncontrado = pAntigo; setorEncontrado = pAntigo.setor || 'corredor'; leitoEncontrado = pAntigo.leito || '01'; dataNascimentoPacienteAtual = pAntigo.dtNasc || ""; break; } } } } } if (pacienteEncontrado && pacienteEncontrado.nome) { if (jaCadastradoHoje) { // Se já foi lançado hoje, exibe apenas a caixinha azul informativa padrão document.getElementById('span-nome').textContent = pacienteEncontrado.nome; document.getElementById('span-setor').textContent = `${setorEncontrado.toUpperCase()} (Leito ${leitoEncontrado})`; let idadeTexto = ""; let ehPed = (setorEncontrado === 'enf-pediatria'); if (dataNascimentoPacienteAtual) { const idadeVal = calcularIdadeAnos(dataNascimentoPacienteAtual); idadeTexto = ` — ${idadeVal} anos`; if (idadeVal < 18) ehPed = true; } document.getElementById('span-idade-info').textContent = idadeTexto; if (infoBox) infoBox.style.display = 'block'; if (novoBox) novoBox.style.display = 'none'; alternarModoFormulario(ehPed); } else { // Se veio de dias atrás, abre a caixa laranja JÁ PREENCHIDA com o nome, data, setor e leito anteriores if (infoBox) infoBox.style.display = 'none'; if (novoBox) { novoBox.style.display = 'block'; document.getElementById('novo-nome-paciente').value = pacienteEncontrado.nome; document.getElementById('novo-dtnasc-paciente').value = dataNascimentoPacienteAtual; document.getElementById('novo-setor-paciente').value = setorEncontrado; document.getElementById('novo-leito-paciente').value = leitoEncontrado; verificarSetorOuIdade(); } } } else { // Se realmente nunca existiu em lugar nenhum, abre limpo para cadastro novo if (infoBox) infoBox.style.display = 'none'; if (novoBox) { document.getElementById('novo-nome-paciente').value = ""; document.getElementById('novo-dtnasc-paciente').value = ""; document.getElementById('novo-setor-paciente').selectedIndex = 0; document.getElementById('novo-leito-paciente').selectedIndex = 0; novoBox.style.display = 'block'; document.getElementById('novo-nome-paciente').focus(); verificarSetorOuIdade(); } } }
